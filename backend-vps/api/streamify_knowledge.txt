MEMORY #2 — REACT HOOKS RULE
================================================================
REACT: Hooks (useState/useEffect) HARUS semua dideklarasikan SEBELUM
conditional return. Urutan wajib: hooks → conditional return → JSX utama.
Hooks setelah return = React error #310 (Rules of Hooks violation).

BENAR:
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  if (loading) return <Spinner />;  ← boleh di sini
  return <div>{data}</div>;

SALAH (error #310):
  const [data, setData] = useState(null);
  if (!data) return null;  ← conditional return dulu
  const [loading, setLoading] = useState(true);  ← hooks SETELAH return = ERROR

================================================================
MEMORY #3 — PHP TIMEOUT RULE
================================================================
PHP: file_get_contents WAJIB pakai timeout:
  $ctx = stream_context_create(['http'=>['timeout'=>15]]);
  $html = file_get_contents($url, false, $ctx);
Tanpa timeout = script stuck selamanya kalau URL lambat/tidak respond.
Recommended timeout: 15 detik (10 untuk scraper cepat).

MEMORY #6 — PHP ACTION HANDLER PATTERN
================================================================
PHP: Setiap action handler HARUS diakhiri exit. Jangan http_response_code() di tengah.
  if ($action === 'list') {
    echo json_encode($result);
    exit;  ← WAJIB
  }

Fungsi helper standar:
  function success($data) { echo json_encode(['success'=>true,'data'=>$data]); exit; }
  function error($msg)    { echo json_encode(['success'=>false,'error'=>$msg]); exit; }

MEMORY #7 — SHELL HEREDOC RULE
================================================================
SHELL: nohup php << 'EOF' TIDAK JALAN (heredoc inline di terminal).
Solusi benar:
  cat > /tmp/script.php << 'PHPEOF'
  <?php // kode PHP
  PHPEOF
  nohup php /tmp/script.php > /var/log/xxx.log 2>&1 &
shell_exec() jalan sebagai www-data. Log file harus chmod 666.

--- STRIP ADS FUNCTION ---

function stripAds($content) {
    // Strip VDO.AI script tags
    $content = preg_replace('/<script[^>]*vdo\.ai[^>]*>.*?<\/script>/si', '', $content);
    $content = preg_replace('/<script[^>]*VideoAd[^>]*>.*?<\/script>/si', '', $content);
    
    // Strip div container iklan
    $content = preg_replace('/<div[^>]*id="[^"]*vdo[^"]*"[^>]*>.*?<\/div>/si', '', $content);
    
    // Strip semua script tags (aman untuk novel content)
    $content = preg_replace('/<script\b[^<]*(?:(?!<\/script>)<[^<]*)*<\/script>/si', '', $content);
    
    return trim($content);
}

    $sep = strpos($path, '?') !== false ? '&' : '?';
    $url = "$MELOLO_BASE$path{$sep}lang=id&code=$MELOLO_CODE";

    $r = @file_get_contents($url, false, $ctx);
    return $r ? json_decode($r, true) : null;

function melolo_list($offset = 0) {
    $d = melolo_fetch("/api/home?offset=$offset");
    if (!$d || $d['code'] !== 0) return null;
    
    $books = [];
    foreach ($d['data']['cell']['cell_data'] ?? [] as $cell) {
        foreach ($cell['books'] ?? [] as $book) {
            // Parse genre dari category_info (JSON string!)
            $genres = [];
            $cat = json_decode($book['category_info'] ?? '[]', true) ?: [];
            foreach ($cat as $c) {
                if ($c['MainCategory'] ?? false) $genres[] = $c['Name'];
            }
            $books[] = [
                'id'       => $book['book_id'],
                'title'    => $book['book_name'],
                'cover'    => $book['thumb_url'],
                'synopsis' => $book['abstract'],
                'episodes' => (int)($book['serial_count'] ?? 0),
                'genres'   => $genres,
                'is_hot'   => ($book['is_hot'] ?? '') === '1',
                'is_new'   => ($book['is_new_book'] ?? '') === '1',
                'source'   => 'melolo',
            ];
        }
    }
    return [
        'content'     => $books,
        'has_more'    => $d['data']['has_more'] ?? false,
        'next_offset' => $d['data']['next_offset'] ?? $offset + 20
    ];
}

        $servers[] = ['label' => $item['definition'], 'url' => $item['url']];

        'backup'  => $d['backup'] ?? '',
        'servers' => $servers,
    ];
}

function stardust_detail($slug, $id) {
    $d = stardust_fetch("/v1/detail/$slug/$id");
    if (!$d) return null;
    $data = $d['data'] ?? [];
    
    // Episodes adalah dict: {'1': {h264, h265}, '2': {...}}
    // BUKAN array biasa!
    $videos = [];
    $episodes_data = $data['episodes'] ?? [];
    foreach ($episodes_data as $ep_num => $val) {
        if (!is_array($val)) continue; // Skip non-episode fields
        $ep_num = (int)$ep_num;
        $url = $val['h264'] ?? $val['h265'] ?? '';
        $servers = [];
        if (!empty($val['h264'])) $servers[] = ['label' => 'H264', 'url' => $val['h264']];
        if (!empty($val['h265'])) $servers[] = ['label' => 'H265', 'url' => $val['h265']];
        $videos[] = [
            'episode' => $ep_num,
            'vid'     => "$slug|$id|$ep_num",  // format untuk video fetch
            'url'     => $url,
            'servers' => $servers,
            'duration'=> 0,
        ];
    }
    // Sort by episode number
    usort($videos, fn($a, $b) => $a['episode'] - $b['episode']);
    
    return [
        'id'       => $id,
        'slug'     => $slug,
        'title'    => $data['title'] ?? '',
        'cover'    => $data['poster'] ?? $data['cover'] ?? '',
        'synopsis' => $data['intro'] ?? '',
        'episodes' => (int)($data['totalEpisodes'] ?? count($videos)),
        'videos'   => $videos,
        'source'   => 'stardusttv',
    ];
}

    $max_eps = (int)($info['maxEps'] ?? $info['maxOffset'] ?? 0);
    $videos = [];
    for ($i = 1; $i <= $max_eps; $i++) {
        $videos[] = [
            'episode' => $i,
            'vid'     => "$id|$i",  // format untuk watch
            'url'     => '',         // URL diambil saat play
            'duration'=> 0,
        ];
    }
    
    return [
        'id'       => $id,
        'title'    => $info['contentName'] ?? $info['shortPlayName'] ?? '',
        'cover'    => $info['contentImg'] ?? $info['shortPlayCover'] ?? '',
        'synopsis' => $info['contentRemark'] ?? '',
        'episodes' => $max_eps,
        'videos'   => $videos,
        'source'   => 'netshort',
    ];
}

function netshort_video($id, $ep) {
    $d = netshort_fetch("/api/watch/$id/$ep");
    if (!$d || !in_array($d['code'] ?? 0, [0, 200])) return null;
    $data = $d['data'] ?? [];
    
    $url = $data['videoUrl'] ?? '';
    if (!$url) return null;
    
    // Cari subtitle Indonesia
    $subtitle = '';
    foreach ($data['subtitles'] ?? [] as $sub) {
        if (($sub['lang'] ?? '') === 'id_ID') {
            $subtitle = $sub['url'];
            break;
        }
    }
    
    return [
        'url'      => $url,
        'subtitle' => $subtitle,
        'servers'  => [['label' => 'Default', 'url' => $url]],
    ];
}

function netshort_search($q, $page = 1) {
    $d = netshort_fetch("/api/search?q=" . urlencode($q) . "&page=$page");
    if (!$d || !in_array($d['code'] ?? 0, [0, 200])) return null;
    
    // Hasil di searchCodeSearchResult, BUKAN simpleSearchResult!
    $items = $d['data']['searchCodeSearchResult'] ?? 
             $d['data']['simpleSearchResult'] ?? [];
    
    $books = [];
    foreach ($items as $item) {
        $books[] = [
            'id'       => $item['shortPlayId'] ?? '',  // pakai shortPlayId!
            'title'    => strip_tags($item['shortPlayName'] ?? ''), // strip <em> tag
            'cover'    => $item['shortPlayCover'] ?? '',
            'synopsis' => '',
            'episodes' => (int)($item['maxEps'] ?? 0),
            'source'   => 'netshort',
        ];
    }
    return ['content' => $books, 'total' => count($books)];
}

<?php

if (!$url) { http_response_code(400); exit; }

$url = preg_replace_callback('/[^\x00-\x7F]/', function($m) {
    return rawurlencode($m[0]);
}, $url);

$ctx = stream_context_create(['http' => [
    'timeout' => 15,
    'header'  => "User-Agent: Mozilla/5.0\r\n"
]]);
$srt = @file_get_contents($url, false, $ctx);

if (!$srt) {
    header('Content-Type: text/vtt');
    echo "WEBVTT\n\n";
    exit;
}

// Convert SRT → WebVTT
// Perbedaan: koma diganti titik di timestamp
// SRT:  00:00:15,866 --> 00:00:18,066
// VTT:  00:00:15.866 --> 00:00:18.066
$vtt = "WEBVTT\n\n";
$vtt .= preg_replace('/(\d{2}:\d{2}:\d{2}),(\d{3})/', '$1.$2', $srt);

header('Content-Type: text/vtt; charset=utf-8');

--- SUBTITLE SYSTEM (VELOLO) ---
Velolo episode punya field "zimu" = URL ke file .srt
Contoh: https://res.velolo.tv/indonesiaPlay/2028390.../纨绔大小姐srt/1.id.srt

Masalah: URL punya karakter China (纨绔大小姐) di path → tidak bisa fetch langsung

Solusi: subtitle-proxy.php
File: /var/www/streamify/app/backend-vps/api/subtitle-proxy.php

Cara kerja subtitle-proxy.php:
1. Terima URL via GET parameter ?url=xxx
2. $_GET sudah auto-decode URL → URL berisi karakter China
3. preg_replace_callback encode karakter non-ASCII:
   preg_replace_callback('/[^\x00-\x7F]/', function($m) {
     return rawurlencode($m[0]);
   }, $url)
4. Fetch file .srt dari velolo dengan URL yang sudah di-encode
5. Convert SRT → WebVTT:
   - Tambah header "WEBVTT\n\n"
   - Ganti koma di timestamp: 00:00:15,866 → 00:00:15.866
6. Return dengan header Content-Type: text/vtt; charset=utf-8

Cara pakai di React (ShortDramaDetailPage.tsx):
  setSubtitleUrl('/api/subtitle-proxy.php?url=' + encodeURIComponent(directSubtitle))
  
  Kemudian render:
  <video>
    <track kind="subtitles" src={subtitleUrl} srcLang="id" label="Indonesia" default />
  </video>

PENTING: encodeURIComponent sekali saja (single encode)!
Double encode tidak bekerja.

function loadVideo(vid, ep, directUrl?, directSubtitle?) {
  if (directUrl) {
    // Velolo, StardustTV, NetShort → URL sudah ada dari detail
    setVideoUrl(directUrl)
    if (directSubtitle) {

    }
    return
  }

    .then(r => r.json())

      setServers(res.data.servers)  // untuk quality selector
    })
}
